const LEVEL_NAMES = {
    hsk1: "HSK 1 · 3.0",
    hsk2: "HSK 2 · 3.0",
    hsk3: "HSK 3 · 3.0"
};

let currentLevel = "hsk1";
let currentIndex = 0;
let isFlipped = false;
let cardOrder = [];

const known = {
    hsk1: loadKnown("hsk1"),
    hsk2: loadKnown("hsk2"),
    hsk3: loadKnown("hsk3")
};

function loadKnown(level) {
    try {
        const saved = JSON.parse(localStorage.getItem(level + "Known") || "[]");
        if (!Array.isArray(saved)) return new Set();
        return new Set(saved.filter(item => Number.isInteger(item) && item >= 0));
    } catch (e) { return new Set(); }
}

function saveKnown(level) {
    try {
        localStorage.setItem(level + "Known", JSON.stringify([...known[level]]));
    } catch (e) { console.warn("Не удалось сохранить прогресс:", e); }
}

function escapeHTML(value) {
    return String(value).replace(/[&<>"']/g, char => ({
        "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#039;"
    })[char]);
}

function getCurrentData() { return DATA[currentLevel] || []; }

function getFilteredIndices() {
    const data = getCurrentData();
    const mode = document.getElementById("cardMode").value;
    return data.map((word, index) => index).filter(index => {
        const isKnown = known[currentLevel].has(index);
        if (mode === "unknown") return !isKnown;
        if (mode === "known") return isKnown;
        return true;
    });
}

function getCurrentWordIndex() { return cardOrder[currentIndex]; }
function getCurrentWord() {
    const index = getCurrentWordIndex();
    if (index === undefined) return null;
    return getCurrentData()[index];
}

function rebuildCardOrder() {
    cardOrder = getFilteredIndices();
    if (currentIndex >= cardOrder.length) currentIndex = Math.max(0, cardOrder.length - 1);
    if (currentIndex < 0) currentIndex = 0;
}

function renderCard() {
    const word = getCurrentWord();
    const hanzi = document.getElementById("cardHanzi");
    const pinyin = document.getElementById("cardPinyin");
    const translation = document.getElementById("cardTranslation");
    const hint = document.getElementById("cardHint");
    const counter = document.getElementById("cardCounter");
    const card = document.getElementById("card");

    if (!word) {
        hanzi.textContent = "🎉";
        pinyin.textContent = "Нет карточек";
        translation.textContent = "";
        hint.textContent = "Измени фильтр или выбери другой уровень.";
        counter.textContent = "0 / 0";
        translation.classList.add("hidden");
        card.classList.remove("flipped");
        ["prevBtn","nextBtn","flipBtn","knowBtn","repeatBtn"].forEach(id => {
            document.getElementById(id).disabled = true;
        });
        return;
    }

    hanzi.textContent = word[0];
    pinyin.textContent = word[1];
    translation.textContent = word[2];
    translation.classList.toggle("hidden", !isFlipped);
    card.classList.toggle("flipped", isFlipped);

    hint.textContent = isFlipped ? "Перевод слова" : "Попробуй вспомнить перевод";
    counter.textContent = `${currentIndex + 1} / ${cardOrder.length}`;

    ["prevBtn","nextBtn"].forEach(id => document.getElementById(id).disabled = cardOrder.length <= 1);
    ["flipBtn","knowBtn","repeatBtn"].forEach(id => document.getElementById(id).disabled = false);

    updateProgress();
}

function flipCard() { if (!getCurrentWord()) return; isFlipped = !isFlipped; renderCard(); }
function nextCard() { if (cardOrder.length === 0) return; currentIndex = (currentIndex + 1) % cardOrder.length; isFlipped = false; renderCard(); }
function prevCard() { if (cardOrder.length === 0) return; currentIndex = (currentIndex - 1 + cardOrder.length) % cardOrder.length; isFlipped = false; renderCard(); }
function shuffleCards() {
    if (cardOrder.length <= 1) return;
    for (let i = cardOrder.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [cardOrder[i], cardOrder[j]] = [cardOrder[j], cardOrder[i]];
    }
    currentIndex = 0; isFlipped = false; renderCard();
}

function updateWordStatus(status) {
    const wordIndex = getCurrentWordIndex();
    if (wordIndex === undefined) return;

    if (status === "known") known[currentLevel].add(wordIndex);
    else known[currentLevel].delete(wordIndex);
    saveKnown(currentLevel);

    const oldLength = cardOrder.length;
    rebuildCardOrder();
    if (cardOrder.length < oldLength && currentIndex >= cardOrder.length) currentIndex = 0;

    isFlipped = false; renderCard(); updateProgress(); renderWordList();
}

function updateProgress() {
    const data = getCurrentData();
    const total = data.length;
    const knownCount = known[currentLevel].size;
    const remaining = Math.max(0, total - knownCount);
    const percent = total ? Math.round(knownCount / total * 100) : 0;

    document.getElementById("progressTitle").textContent = LEVEL_NAMES[currentLevel];
    document.getElementById("percent").textContent = percent + "%";
    document.getElementById("progressBar").style.width = percent + "%";
    document.getElementById("knownCount").textContent = knownCount;
    document.getElementById("totalCount").textContent = total;
    document.getElementById("remainingCount").textContent = remaining;

    const allLevels = ["hsk1", "hsk2", "hsk3"];
    let totalAll = 0, knownAll = 0;
    allLevels.forEach(level => {
        totalAll += DATA[level].length;
        knownAll += known[level].size;
    });
    const overallPercent = totalAll ? Math.round(knownAll / totalAll * 100) : 0;
    document.getElementById("overallPercent").textContent = overallPercent + "%";
    document.getElementById("overallBar").style.width = overallPercent + "%";
    document.getElementById("overallText").textContent = `${knownAll} из ${totalAll} слов выучено`;
}

function renderWordList() {
    const container = document.getElementById("wordList");
    const query = document.getElementById("search").value.trim().toLowerCase();
    const mode = document.getElementById("trackerMode").value;
    const data = getCurrentData();

    const filtered = data.map((word, index) => ({ word, index })).filter(item => {
        const isKnown = known[currentLevel].has(item.index);
        if (mode === "known" && !isKnown) return false;
        if (mode === "unknown" && isKnown) return false;
        return item.word.join(" ").toLowerCase().includes(query);
    });

    if (filtered.length === 0) {
        container.innerHTML = `<div class="empty-state"><div style="font-size:38px;margin-bottom:10px">🔎</div><strong>Ничего не найдено</strong><p>Попробуй изменить запрос или фильтр.</p></div>`;
        return;
    }

    container.innerHTML = filtered.map(({ word, index }) => {
        const isKnown = known[currentLevel].has(index);
        return `
        <label class="word-item ${isKnown ? "known" : ""}">
            <input type="checkbox" data-word-index="${index}" ${isKnown ? "checked" : ""}>
            <span class="word-hanzi">${escapeHTML(word[0])}</span>
            <span class="word-details">
                <span class="word-pinyin">${escapeHTML(word[1])}</span>
                <span class="word-translation">${escapeHTML(word[2])}</span>
            </span>
        </label>`;
    }).join("");
}

function changeLevel(level) {
    if (!DATA[level]) return;
    currentLevel = level;
    currentIndex = 0;
    isFlipped = false;

    document.querySelectorAll("[data-level]").forEach(button => {
        button.classList.toggle("active", button.dataset.level === level);
    });

    document.getElementById("levelBadge").textContent = LEVEL_NAMES[level];
    rebuildCardOrder();
    renderCard();
    renderWordList();
    updateProgress();
    document.getElementById("trackerTitle").textContent = `📚 Мой словарь — ${LEVEL_NAMES[level]}`;
}

function changePage(page) {
    document.querySelectorAll("[data-page]").forEach(button => {
        button.classList.toggle("active", button.dataset.page === page);
    });
    document.getElementById("cardsPage").classList.toggle("hidden", page !== "cards");
    document.getElementById("trackerPage").classList.toggle("hidden", page !== "tracker");

    if (page === "tracker") renderWordList();
}

document.querySelectorAll("[data-level]").forEach(button => {
    button.addEventListener("click", () => changeLevel(button.dataset.level));
});

document.querySelectorAll("[data-page]").forEach(button => {
    button.addEventListener("click", () => changePage(button.dataset.page));
